/** * Kotoba-cho API (Phase 1) * Back end for Pete's Japanese notebook. }); return entries.reverse(); } /* ---------- writing ---------- */ function addWord_(word) { if (!word || !String(word.jp || '').trim() || !String(word.en || '').trim()) { throw new Error('Japanese and English meaning are required.'); } const sh = sheet_(VOCAB); const row = lastDataRow_(sh, 2) + 1; // after the last word; rank formulas in P:S stay untouched const values = VOCAB_FIELDS.map(k => k === 'date' ? today_() : cell_(word[k] || (k === 'mastery' ? return start; } /* ---------- generic range access for the notebook app ---------- */ function rangeOf_(a1) { const s = String(a1 || ''); const i = s.lastIndexOf('!'); if (i < 0) throw new Error('Range needs a tab name, like "Vocab Bank!A1:O10".'); } /* ---------- Phase 4: Claude through your own API key ---------- */ function aiKey_() { return PropertiesService.getScriptProperties().getProperty('ANTHROPIC_KEY'); } function ai_(p) { const key = aiKey_(); if (!key) return { ok: false, error: 'no-ai-key' }; const prompt = String(p.prompt || ''); if (!prompt) throw new Error('Empty prompt.'); if (prompt.length > 12000) throw new Error('Prompt too long.'); const payload = { model: AI_MODEL, max_tokens: Math.min(Number(p.max) || 1000, 1500), messages: [{ role: 'user', content: prompt }] }; if (p.json) payload.system = 'Return only valid JSON. No prose, no markdown, no code fences.'; const res = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', { method: 'post', contentType: 'application/json', headers: { 'x-api-key': key, 'anthropic-version': '2023-06-01' }, payload: JSON.stringify(payload), muteHttpExceptions: true }); const code = res.getResponseCode(); let body = {}; try { body = JSON.parse(res.getContentText() || '{}'); } catch (e) {} if (code === 429 || code === 529) return { ok: false, error: 'rate_limited' }; if (code >= 400) return { ok: false, error: 'ai-error: ' + ((body.error && body.error.message) || ('HTTP ' + code)) }; const text = (body.content || []).filter(b => b.type === 'text').map(b => b.text).join(''); return { ok: true, text: text }; } // Run this once by hand after adding your key. It asks Google for permission // to reach the Anthropic API and checks the key works. function testAI() { const r = ai_({ prompt: 'Reply with one short friendly Japanese greeting.' }); Logger.log(JSON.stringify(r)); } /* ---------- helpers ---------- */ function sheet_(name) { const sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(name); if (!sh) throw new Error('Missing tab: ' + name); return sh; } // Last row that has a value in the given columns (ignores formula-only rows). function lastDataRow_(sh, firstCol, numCols) { const max = sh.getLastRow(); if (max < 1) return 1; const vals = sh.getRange(1, firstCol, max, numCols || 1).getValues(); for (let i = vals.length - 1; i >= 0; i--) { if (vals[i].some(v => String(v).trim() !== '')) return i + 1; } return 1; } function wordLink_(w, gid) { const q = String(w).replace(/"/g, '""'); return '=LET(w,"' + q + '",IFERROR(HYPERLINK("#gid=' + gid + '&range=A"&MATCH(w,\'Vocab Bank\'!$B:$B,0)&":M"&MATCH(w,\'Vocab Bank\'!$B:$B,0),w),w))'; } function cell_(v) { v = String(v == null ?
